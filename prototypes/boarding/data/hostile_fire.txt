# How accurately anything hostile shoots at the player (guards, turrets, everything).
#
# Damage is never reduced or capped. Fairness comes from aim instead:
#   1. Warning shots: an engagement opens with deliberately wild shots that are guaranteed
#      to miss, landing close enough to see.
#   2. Accuracy ramp: after the warnings, the spread cone shrinks to the baseline.
#   3. Breaking line of sight resets both.
# Spread is the full cone angle in degrees.

[ramp]
warning_shots = 3          # the first shots at a target always miss...
warning_seconds = 0.75     # ...and so does everything fired in the first 0.75 s
warning_miss_min = 1.0     # metres outside the target's outline that a warning shot lands...
warning_miss_max = 2.4     # ...up to this far
ramp_seconds = 2.0         # after the warnings, spread shrinks to the baseline over this long
ramp_start = 3.0           # starting from this many times the baseline
los_reset_seconds = 0.4    # out of sight this long and the next engagement starts over

[ratings]
very_good = 0.4
good = 1.2
fair = 2.5
poor = 4.5
very_poor = 8.0

# Shooter class: its rating against on_foot, hover, small_vessel, capital (or base).
[baseline]
infantry = good, fair, poor, very_poor
small_craft = fair, good, good, good
capital = very_poor, poor, fair, very_good

# Situations that widen the cone (multipliers).
[conditions]
target_sprinting = 1.4
target_crouched = 1.15
lights_out = 1.4
